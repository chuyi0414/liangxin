# 02. 数据配置系统

> 本章教你：用 ScriptableObject 创建可在 Inspector 中编辑的配置表

---

## 🎯 本章目标

1. 理解 ScriptableObject 是什么
2. 创建单位配置（友方、敌方）
3. 创建词条配置
4. 在 Unity 编辑器中配置数据

完成后，你可以：
- 在 Inspector 中直接编辑单位属性
- 不用改代码就能调整数值
- 方便策划配置游戏数据

---

## 📖 概念讲解：什么是 ScriptableObject？

**问题**：游戏中有 50 种友方单位、141 种敌人，每个都有名字、血量、攻击力...这些数据放哪？

**方案 A：写死在代码里（❌ 不好）**
```csharp
// 每次改数值都要改代码，容易出错
public class Rider : Unit
{
    public int hp = 100;
    public int attack = 20;
    // 改一个数字就要重新编译...
}
```

**方案 B：用 ScriptableObject（✅ 推荐）**
```csharp
// 数据和代码分离，在 Inspector 中编辑
[CreateAssetMenu(fileName = "NewUnit", menuName = "良心防线/单位配置")]
public class UnitConfig : ScriptableObject
{
    public string unitName;
    public int maxHP;
    public int attack;
}
```

**ScriptableObject 的好处**：
1. 数据保存为 `.asset` 文件
2. 在 Inspector 中直接编辑
3. 不用改代码
4. 支持版本控制
5. 运行时共享同一份数据（省内存）

## 📝 第一步：创建单位基础配置

### 1.1 创建 UnitConfig 脚本

1. 右键 `Assets/_Game/Scripts/Data`
2. `Create > C# Script`
3. 命名为 `UnitConfig`

### 1.2 编写代码

```csharp
using UnityEngine;

namespace LiangXin.Data
{
    /// <summary>
    /// 单位基础配置
    /// 友方和敌方单位共用的基础属性
    /// </summary>
    [CreateAssetMenu(fileName = "NewUnit", menuName = "良心防线/单位配置/基础单位")]
    public class UnitConfig : ScriptableObject
    {
        [Header("基础信息")]
        [Tooltip("单位唯一 ID，如 F01, E01")]
        public string unitId;
        
        [Tooltip("单位名称")]
        public string unitName;
        
        [Tooltip("单位描述")]
        [TextArea(2, 5)]
        public string description;
        
        [Tooltip("单位图标")]
        public Sprite icon;
        
        [Header("基础属性")]
        [Tooltip("最大生命值")]
        [Range(1, 10000)]
        public int maxHP = 100;
        
        [Tooltip("攻击力")]
        [Range(0, 1000)]
        public int attack = 10;
        
        [Tooltip("防御力")]
        [Range(0, 500)]
        public int defense = 0;
        
        [Tooltip("攻击速度（每秒攻击次数）")]
        [Range(0.1f, 10f)]
        public float attackSpeed = 1f;
        
        [Tooltip("移动速度")]
        [Range(0f, 20f)]
        public float moveSpeed = 5f;
        
        [Tooltip("攻击范围")]
        [Range(0.5f, 50f)]
        public float attackRange = 2f;
        
        [Header("预制体")]
        [Tooltip("单位预制体")]
        public GameObject prefab;
    }
}
```

## 📝 第二步：创建员工配置（继承 UnitConfig）

### 2.1 创建 EmployeeConfig 脚本

1. 右键 `Assets/_Game/Scripts/Data`
2. `Create > C# Script`
3. 命名为 `EmployeeConfig`

### 2.2 编写代码

```csharp
using UnityEngine;
using System.Collections.Generic;

namespace LiangXin.Data
{
    /// <summary>
    /// 员工（友方单位）配置
    /// 继承自 UnitConfig，添加员工特有属性
    /// </summary>
    [CreateAssetMenu(fileName = "NewEmployee", menuName = "良心防线/单位配置/员工")]
    public class EmployeeConfig : UnitConfig
    {
        [Header("======== 员工信息 ========")]
        
        [Tooltip("所属部门")]
        public DepartmentType department;
        
        [Tooltip("招募费用")]
        [Range(0, 1000)]
        public int recruitCost = 50;
        
        [Tooltip("升级费用（Lv1→2, Lv2→3, ...）")]
        public int[] upgradeCosts = new int[] { 0, 50, 100, 150, 200 };
        
        [Header("等级属性加成")]
        [Tooltip("每级血量加成百分比")]
        [Range(0, 100)]
        public int hpBonusPerLevel = 10;
        
        [Tooltip("每级攻击加成百分比")]
        [Range(0, 100)]
        public int attackBonusPerLevel = 15;
        
        [Header("Lv1 状态：正式员工（残缺状态）")]
        [TextArea(2, 4)]
        public string lv1Description;
        
        [Header("Lv2 状态：资深员工（完全体）")]
        [TextArea(2, 4)]
        public string lv2Description;
        
        [Tooltip("Lv2 自动技能冷却时间")]
        [Range(0.5f, 30f)]
        public float lv2SkillCooldown = 3f;
        
        [Header("Lv5 状态：传奇员工")]
        [TextArea(2, 4)]
        public string lv5Description;
        
        [Tooltip("传奇任务描述")]
        public string legendaryMission;
        
        [Tooltip("传奇大招冷却时间")]
        [Range(10f, 300f)]
        public float ultimateCooldown = 60f;
        
        [Header("词条池")]
        [Tooltip("可能携带的负面词条 ID 列表")]
        public List<string> negativeTrait;
        
        [Tooltip("可能获得的正面词条 ID 列表")]
        public List<string> positiveTraits;
        
        /// <summary>
        /// 获取指定等级的升级费用
        /// </summary>
        public int GetUpgradeCost(int currentLevel)
        {
            if (currentLevel < 1 || currentLevel >= upgradeCosts.Length)
                return 0;
            return upgradeCosts[currentLevel];
        }
        
        /// <summary>
        /// 计算指定等级的最终血量
        /// </summary>
        public int GetHPAtLevel(int level)
        {
            float bonus = 1f + (level - 1) * hpBonusPerLevel / 100f;
            return Mathf.RoundToInt(maxHP * bonus);
        }
        
        /// <summary>
        /// 计算指定等级的最终攻击力
        /// </summary>
        public int GetAttackAtLevel(int level)
        {
            float bonus = 1f + (level - 1) * attackBonusPerLevel / 100f;
            return Mathf.RoundToInt(attack * bonus);
        }
    }
    
    /// <summary>
    /// 部门类型
    /// </summary>
    public enum DepartmentType
    {
        CityFoundation,     // 城市基石（民生与服务）
        Industrial,         // 工业与基建
        Professional,       // 专业技术
        Cultural,           // 文体艺术
        Special             // 特殊工种
    }
}
```

## 📝 第四步：创建词条配置

### 4.1 创建 TraitConfig 脚本

1. 右键 `Assets/_Game/Scripts/Data`
2. `Create > C# Script`
3. 命名为 `TraitConfig`

### 4.2 编写代码

```csharp
using UnityEngine;
using System.Collections.Generic;

namespace LiangXin.Data
{
    /// <summary>
    /// 词条配置
    /// </summary>
    [CreateAssetMenu(fileName = "NewTrait", menuName = "良心防线/词条")]
    public class TraitConfig : ScriptableObject
    {
        [Header("基础信息")]
        [Tooltip("词条 ID，如 T-N01, T-P01")]
        public string traitId;
        
        [Tooltip("词条名称")]
        public string traitName;
        
        [Tooltip("词条类型")]
        public TraitType traitType;
        
        [Tooltip("效果描述")]
        [TextArea(2, 4)]
        public string effectDescription;
        
        [Tooltip("隐喻/设计意图")]
        [TextArea(2, 4)]
        public string metaphor;
        
        [Tooltip("词条图标")]
        public Sprite icon;
        
        [Header("效果参数")]
        [Tooltip("效果列表")]
        public List<TraitEffect> effects = new List<TraitEffect>();
        
        [Header("触发条件")]
        [Tooltip("触发类型")]
        public TraitTriggerType triggerType = TraitTriggerType.Passive;
        
        [Tooltip("触发条件描述")]
        public string triggerCondition;
    }
    
    /// <summary>
    /// 词条类型
    /// </summary>
    public enum TraitType
    {
        Negative,   // 负面词条
        Positive,   // 正面词条
        Legendary   // 传奇词条（职业专属）
    }
    
    /// <summary>
    /// 词条触发类型
    /// </summary>
    public enum TraitTriggerType
    {
        Passive,        // 被动：始终生效
        OnHit,          // 命中时
        OnBeingHit,     // 被击中时
        OnKill,         // 击杀时
        OnSkillUse,     // 释放技能时
        OnLowHP,        // 低血量时
        Periodic,       // 周期性触发
        Conditional     // 条件触发
    }
    
    /// <summary>
    /// 词条效果
    /// </summary>
    [System.Serializable]
    public class TraitEffect
    {
        [Tooltip("效果类型")]
        public TraitEffectType effectType;
        
        [Tooltip("目标属性")]
        public StatType targetStat;
        
        [Tooltip("修改类型")]
        public ModifierType modifierType;
        
        [Tooltip("数值")]
        public float value;
        
        [Tooltip("持续时间（0 表示永久）")]
        public float duration;
        
        [Tooltip("附加条件")]
        public string condition;
    }
    
    /// <summary>
    /// 效果类型
    /// </summary>
    public enum TraitEffectType
    {
        ModifyStat,         // 修改属性
        TriggerAction,      // 触发动作（如眩晕、位移）
        SpecialBehavior     // 特殊行为
    }
    
    /// <summary>
    /// 属性类型
    /// </summary>
    public enum StatType
    {
        MaxHP,
        CurrentHP,
        Attack,
        Defense,
        AttackSpeed,
        MoveSpeed,
        AttackRange,
        CritRate,
        CritDamage,
        DamageReduction,
        Lifesteal
    }
    
    /// <summary>
    /// 修改器类型
    /// </summary>
    public enum ModifierType
    {
        Flat,       // 固定值加减
        Percent,    // 百分比加减
        Override    // 覆盖
    }
}
```

## 📝 第六步：创建配置数据库

为了方便运行时查找配置，我们创建一个配置数据库：

### 6.1 创建 ConfigDatabase 脚本

1. 右键 `Assets/_Game/Scripts/Data`
2. `Create > C# Script`
3. 命名为 `ConfigDatabase`

### 6.2 编写代码

```csharp
using UnityEngine;
using System.Collections.Generic;

namespace LiangXin.Data
{
    /// <summary>
    /// 配置数据库
    /// 存储所有游戏配置的引用
    /// </summary>
    [CreateAssetMenu(fileName = "ConfigDatabase", menuName = "良心防线/配置数据库")]
    public class ConfigDatabase : ScriptableObject
    {
        [Header("员工配置")]
        public List<EmployeeConfig> employees = new List<EmployeeConfig>();
        
        [Header("敌人配置")]
        public List<EnemyConfig> enemies = new List<EnemyConfig>();
        
        [Header("词条配置")]
        public List<TraitConfig> traits = new List<TraitConfig>();
        
        // 运行时缓存
        private Dictionary<string, EmployeeConfig> _employeeDict;
        private Dictionary<string, EnemyConfig> _enemyDict;
        private Dictionary<string, TraitConfig> _traitDict;
        
        /// <summary>
        /// 初始化字典缓存
        /// </summary>
        public void Initialize()
        {
            // 员工字典
            _employeeDict = new Dictionary<string, EmployeeConfig>();
            foreach (var config in employees)
            {
                if (config != null && !string.IsNullOrEmpty(config.unitId))
                {
                    _employeeDict[config.unitId] = config;
                }
            }
            
            // 敌人字典
            _enemyDict = new Dictionary<string, EnemyConfig>();
            foreach (var config in enemies)
            {
                if (config != null && !string.IsNullOrEmpty(config.unitId))
                {
                    _enemyDict[config.unitId] = config;
                }
            }
            
            // 词条字典
            _traitDict = new Dictionary<string, TraitConfig>();
            foreach (var config in traits)
            {
                if (config != null && !string.IsNullOrEmpty(config.traitId))
                {
                    _traitDict[config.traitId] = config;
                }
            }
            
            Debug.Log($"[ConfigDatabase] 初始化完成: 员工 {_employeeDict.Count}, 敌人 {_enemyDict.Count}, 词条 {_traitDict.Count}");
        }
        
        /// <summary>
        /// 获取员工配置
        /// </summary>
        public EmployeeConfig GetEmployee(string id)
        {
            if (_employeeDict == null) Initialize();
            return _employeeDict.TryGetValue(id, out var config) ? config : null;
        }
        
        /// <summary>
        /// 获取敌人配置
        /// </summary>
        public EnemyConfig GetEnemy(string id)
        {
            if (_enemyDict == null) Initialize();
            return _enemyDict.TryGetValue(id, out var config) ? config : null;
        }
        
        /// <summary>
        /// 获取词条配置
        /// </summary>
        public TraitConfig GetTrait(string id)
        {
            if (_traitDict == null) Initialize();
            return _traitDict.TryGetValue(id, out var config) ? config : null;
        }
        
        /// <summary>
        /// 获取随机员工
        /// </summary>
        public EmployeeConfig GetRandomEmployee()
        {
            if (employees.Count == 0) return null;
            return employees[Random.Range(0, employees.Count)];
        }
        
        /// <summary>
        /// 获取随机负面词条
        /// </summary>
        public TraitConfig GetRandomNegativeTrait()
        {
            var negativeTraits = traits.FindAll(t => t.traitType == TraitType.Negative);
            if (negativeTraits.Count == 0) return null;
            return negativeTraits[Random.Range(0, negativeTraits.Count)];
        }
        
        /// <summary>
        /// 获取随机正面词条
        /// </summary>
        public TraitConfig GetRandomPositiveTrait()
        {
            var positiveTraits = traits.FindAll(t => t.traitType == TraitType.Positive);
            if (positiveTraits.Count == 0) return null;
            return positiveTraits[Random.Range(0, positiveTraits.Count)];
        }
    }
}
```
