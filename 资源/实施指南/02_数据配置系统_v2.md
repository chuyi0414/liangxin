## 📖 概念讲解：数据表系统

**为什么需要数据表？**

- 策划用 Excel 配置数据
- 程序用代码读取数据
- 数据和代码分离，易于调整

```
Excel 配置表              CY.Data 系统               游戏使用
┌──────────┐            ┌──────────────┐           ┌──────────┐
│ 怪物.csv │  ───→     │ MonsterRow   │   ───→   │ 生成怪物  │
│ 员工.csv │            │ EmployeeRow  │           │ 招募员工  │
│ 波次.csv │            │ WaveRow      │           │ 波次逻辑  │
└──────────┘            └──────────────┘           └──────────┘
```

```csharp
// Assets/_Game/Scripts/Config/MonsterRow.cs
using CYFramework.Core.DataTable;

namespace LiangXin.Config
{
    /// <summary>
    /// 怪物配置
    /// CSV 格式: Id,Name,Hp,Speed,Damage,GoldReward
    /// </summary>
    public class MonsterRow : IDataRow
    {
        public int Id { get; private set; }
        public string Name { get; private set; }
        public int Hp { get; private set; }
        public float Speed { get; private set; }
        public int Damage { get; private set; }
        public int GoldReward { get; private set; }
        
        public void ParseRow(string[] values)
        {
            Id = int.Parse(values[0]);
            Name = values[1];
            Hp = int.Parse(values[2]);
            Speed = float.Parse(values[3]);
            Damage = int.Parse(values[4]);
            GoldReward = int.Parse(values[5]);
        }
    }
}
```

### 1.4 WaveRow.cs

```csharp
// Assets/_Game/Scripts/Config/WaveRow.cs
using CYFramework.Core.DataTable;

namespace LiangXin.Config
{
    /// <summary>
    /// 波次配置
    /// CSV 格式: Id,MonsterIds,MonsterCounts,SpawnInterval,PrepareTime
    /// </summary>
    public class WaveRow : IDataRow
    {
        public int Id { get; private set; }
        public int[] MonsterIds { get; private set; }
        public int[] MonsterCounts { get; private set; }
        public float SpawnInterval { get; private set; }
        public float PrepareTime { get; private set; }
        
        public void ParseRow(string[] values)
        {
            Id = int.Parse(values[0]);
            
            // 解析数组格式 "1001|1002|1003"
            var ids = values[1].Split('|');
            MonsterIds = new int[ids.Length];
            for (int i = 0; i < ids.Length; i++)
            {
                MonsterIds[i] = int.Parse(ids[i]);
            }
            
            var counts = values[2].Split('|');
            MonsterCounts = new int[counts.Length];
            for (int i = 0; i < counts.Length; i++)
            {
                MonsterCounts[i] = int.Parse(counts[i]);
            }
            
            SpawnInterval = float.Parse(values[3]);
            PrepareTime = float.Parse(values[4]);
        }
    }
}
```

## 📝 第三步：加载配置数据

### 3.1 创建 ConfigManager.cs

```csharp
// Assets/_Game/Scripts/Core/ConfigManager.cs
using CYFramework;
using CYFramework.Core.DataTable;
using CYFramework.Infrastructure;
using LiangXin.Config;
using UnityEngine;

namespace LiangXin
{
    /// <summary>
    /// 配置管理器
    /// </summary>
    public static class ConfigManager
    {
        public static bool IsLoaded { get; private set; }
        
        /// <summary>
        /// 加载所有配置表
        /// </summary>
        public static void LoadAll()
        {
            if (IsLoaded) return;
            
            CYLog.Info("[ConfigManager] 开始加载配置...");
            
            // 加载怪物配置
            var monsterCsv = Resources.Load<TextAsset>("Config/Monster");
            if (monsterCsv != null)
            {
                CY.Data.LoadCsv<MonsterRow>(monsterCsv.text, "Monster");
            }
            
            // 加载员工配置
            var employeeCsv = Resources.Load<TextAsset>("Config/Employee");
            if (employeeCsv != null)
            {
                CY.Data.LoadCsv<EmployeeRow>(employeeCsv.text, "Employee");
            }
            
            // 加载波次配置
            var waveCsv = Resources.Load<TextAsset>("Config/Wave");
            if (waveCsv != null)
            {
                CY.Data.LoadCsv<WaveRow>(waveCsv.text, "Wave");
            }
            
            IsLoaded = true;
            CYLog.Info("[ConfigManager] 配置加载完成");
        }
        
        // ==================== 便捷访问方法 ====================
        
        public static MonsterRow GetMonster(int id)
        {
            return CY.Data.GetRow<MonsterRow>(id, "Monster");
        }
        
        public static EmployeeRow GetEmployee(int id)
        {
            return CY.Data.GetRow<EmployeeRow>(id, "Employee");
        }
        
        public static WaveRow GetWave(int waveNumber)
        {
            return CY.Data.GetRow<WaveRow>(waveNumber, "Wave");
        }
        
        public static DataTable<MonsterRow> GetMonsterTable()
        {
            return CY.Data.GetTable<MonsterRow>("Monster");
        }
        
        public static DataTable<EmployeeRow> GetEmployeeTable()
        {
            return CY.Data.GetTable<EmployeeRow>("Employee");
        }
    }
}
```

## 📝 第四步：使用配置数据

### 4.1 获取单条数据

```csharp
// 获取 ID 为 1001 的怪物
var monster = ConfigManager.GetMonster(1001);
Debug.Log($"怪物: {monster.Name}, HP: {monster.Hp}");

// 获取 ID 为 2001 的员工
var employee = ConfigManager.GetEmployee(2001);
Debug.Log($"员工: {employee.Name}, 招募费用: {employee.HireCost}");

// 获取第 3 波配置
var wave = ConfigManager.GetWave(3);
Debug.Log($"波次 3: 怪物数量 {wave.MonsterCounts.Length} 种");
```

### 4.2 条件查询

```csharp
// 获取所有 HP > 500 的怪物（BOSS 级）
var bossMonsters = CY.Data.GetTable<MonsterRow>("Monster")
    .GetRows(m => m.Hp > 500);

foreach (var boss in bossMonsters)
{
    Debug.Log($"BOSS: {boss.Name}");
}

// 获取招募费用 <= 25 的员工（便宜员工）
var cheapEmployees = CY.Data.GetTable<EmployeeRow>("Employee")
    .GetRows(e => e.HireCost <= 25);
```

## ✅ 本章小结

| API | 说明 |
|-----|------|
| `CY.Data.LoadCsv<T>(csv)` | 从 CSV 加载数据表 |
| `CY.Data.GetRow<T>(id)` | 获取单行数据 |
| `CY.Data.GetTable<T>()` | 获取整个数据表 |
| `table.GetRows(predicate)` | 条件查询多行 |
