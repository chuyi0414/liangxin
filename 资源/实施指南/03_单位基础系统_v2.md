## 📖 概念讲解：Entity 实体系统

**为什么用 Entity 系统？**

| 传统方式 | CY.Entity |
|---------|-----------|
| 手动 Instantiate/Destroy | 自动对象池管理 |
| 各种 Manager 分散管理 | 统一 EntityManager |
| 生命周期不清晰 | 明确的 Show/Hide/Update |

```
┌─────────────────────────────────────────────────────────────┐
│                     Entity 实体系统                          │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  注册实体类型                      显示实体                   │
│  CY.Entity.Register("Enemy",     CY.Entity.Show("Enemy",    │
│      enemyPrefab, 20);           new EnemyData{...});       │
│         ↓                               ↓                   │
│  ┌──────────────┐                ┌──────────────┐           │
│  │  对象池预热   │  ──────→      │ 从池中取出    │           │
│  │  创建 20 个   │               │ OnShow()     │           │
│  └──────────────┘                └──────────────┘           │
│                                         │                   │
│                                         ▼                   │
│                                  ┌──────────────┐           │
│  隐藏实体                         │  每帧更新     │           │
│  CY.Entity.Hide(enemy);          │ OnUpdate()   │           │
│         ↓                        └──────────────┘           │
│  ┌──────────────┐                       │                   │
│  │  回收到池中   │  ←────────────────────┘                   │
│  │  OnHide()    │                                           │
│  └──────────────┘                                           │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 1.2 UnitBase.cs

```csharp
// Assets/_Game/Scripts/Entities/UnitBase.cs
using CYFramework;
using CYFramework.Core.Entity;
using CYFramework.Infrastructure;
using UnityEngine;

namespace LiangXin.Entities
{
    /// <summary>
    /// 单位数据
    /// </summary>
    public struct UnitData
    {
        public int ConfigId;
        public Vector3 SpawnPosition;
        public int Level;
    }
    
    /// <summary>
    /// 单位基类
    /// </summary>
    public abstract class UnitBase : EntityBase
    {
        // ==================== 属性 ====================
        
        public int ConfigId { get; protected set; }
        public int MaxHp { get; protected set; }
        public int CurrentHp { get; protected set; }
        public int Damage { get; protected set; }
        public float Speed { get; protected set; }
        public bool IsDead => CurrentHp <= 0;
        
        // ==================== 组件 ====================
        
        protected Rigidbody2D Rb { get; private set; }
        protected SpriteRenderer Sprite { get; private set; }
        
        // ==================== 生命周期 ====================
        
        protected virtual void Awake()
        {
            Rb = GetComponent<Rigidbody2D>();
            Sprite = GetComponent<SpriteRenderer>();
        }
        
        protected override void OnEntityShow(object userData)
        {
            if (userData is UnitData data)
            {
                ConfigId = data.ConfigId;
                transform.position = data.SpawnPosition;
                
                // 子类实现具体初始化
                InitFromConfig(data);
            }
            
            CurrentHp = MaxHp;
        }
        
        protected override void OnEntityHide()
        {
            // 重置状态
            CurrentHp = 0;
        }
        
        protected override void OnEntityUpdate(float deltaTime)
        {
            if (IsDead) return;
            
            // 子类实现具体更新逻辑
            UpdateUnit(deltaTime);
        }
        
        // ==================== 抽象方法 ====================
        
        /// <summary>
        /// 从配置初始化
        /// </summary>
        protected abstract void InitFromConfig(UnitData data);
        
        /// <summary>
        /// 更新逻辑
        /// </summary>
        protected abstract void UpdateUnit(float deltaTime);
        
        // ==================== 公共方法 ====================
        
        /// <summary>
        /// 受到伤害
        /// </summary>
        public virtual void TakeDamage(int damage, UnitBase attacker)
        {
            if (IsDead) return;
            
            CurrentHp -= damage;
            CYLog.Debug($"[{EntityType}] {name} 受到 {damage} 伤害, 剩余 HP: {CurrentHp}");
            
            if (CurrentHp <= 0)
            {
                CurrentHp = 0;
                OnDeath(attacker);
            }
        }
        
        /// <summary>
        /// 死亡处理
        /// </summary>
        protected virtual void OnDeath(UnitBase killer)
        {
            CYLog.Debug($"[{EntityType}] {name} 死亡");
            
            // 发送死亡事件
            var evt = new UnitDiedEvent
            {
                UnitId = Id,
                IsEnemy = this is EnemyEntity,
                KillerId = killer?.Id ?? 0,
                Position = transform.position
            };
            CY.Event.Fire(ref evt);
            
            // 延迟隐藏（播放死亡动画）
            CY.Timer.Delay(0.5f, () => CY.Entity.Hide(this));
        }
    }
}
```

```csharp
// Assets/_Game/Scripts/Entities/EnemyEntity.cs
using CYFramework;
using CYFramework.Infrastructure;
using LiangXin.Config;
using UnityEngine;

namespace LiangXin.Entities
{
    /// <summary>
    /// 敌人实体
    /// </summary>
    public class EnemyEntity : UnitBase
    {
        public override string EntityType => "Enemy";
        
        public int GoldReward { get; private set; }
        
        private Transform _target;  // 目标（老板）
        
        protected override void InitFromConfig(UnitData data)
        {
            var config = ConfigManager.GetMonster(data.ConfigId);
            if (config == null)
            {
                CYLog.Error($"[Enemy] 找不到配置: {data.ConfigId}");
                return;
            }
            
            MaxHp = config.Hp;
            Damage = config.Damage;
            Speed = config.Speed;
            GoldReward = config.GoldReward;
            
            // 查找目标
            _target = GameObject.FindWithTag("Boss")?.transform;
        }
        
        protected override void UpdateUnit(float deltaTime)
        {
            if (_target == null) return;
            
            // 向目标移动
            Vector3 direction = (_target.position - transform.position).normalized;
            transform.position += direction * Speed * deltaTime;
            
            // 面向目标
            if (direction.x != 0)
            {
                Sprite.flipX = direction.x < 0;
            }
            
            // 检测攻击范围
            float distance = Vector3.Distance(transform.position, _target.position);
            if (distance < 1f)
            {
                Attack();
            }
        }
        
        private void Attack()
        {
            // TODO: 攻击逻辑
        }
        
        protected override void OnDeath(UnitBase killer)
        {
            // 给予金币奖励
            LiangXinGame.Game.Resources.AddGold(GoldReward);
            
            base.OnDeath(killer);
        }
    }
}
```

## 📝 第三步：创建员工实体

### 3.1 EmployeeEntity.cs

```csharp
// Assets/_Game/Scripts/Entities/EmployeeEntity.cs
using CYFramework;
using CYFramework.Infrastructure;
using LiangXin.Config;
using UnityEngine;

namespace LiangXin.Entities
{
    /// <summary>
    /// 员工实体
    /// </summary>
    public class EmployeeEntity : UnitBase
    {
        public override string EntityType => "Employee";
        
        public int Level { get; private set; } = 1;
        public CommandType CurrentCommand { get; private set; } = CommandType.Follow;
        
        private Transform _bossTransform;
        private EnemyEntity _targetEnemy;
        private float _attackCooldown;
        private const float ATTACK_INTERVAL = 1f;
        
        protected override void InitFromConfig(UnitData data)
        {
            var config = ConfigManager.GetEmployee(data.ConfigId);
            if (config == null)
            {
                CYLog.Error($"[Employee] 找不到配置: {data.ConfigId}");
                return;
            }
            
            Level = data.Level;
            MaxHp = config.BaseHp * Level;
            Damage = config.BaseDamage * Level;
            Speed = 3f;
            
            _bossTransform = GameObject.FindWithTag("Boss")?.transform;
        }
        
        protected override void UpdateUnit(float deltaTime)
        {
            _attackCooldown -= deltaTime;
            
            switch (CurrentCommand)
            {
                case CommandType.Follow:
                    FollowBoss(deltaTime);
                    break;
                case CommandType.AttackMove:
                    AttackMove(deltaTime);
                    break;
                case CommandType.Hold:
                    HoldPosition(deltaTime);
                    break;
            }
        }
        
        private void FollowBoss(float deltaTime)
        {
            if (_bossTransform == null) return;
            
            float distance = Vector3.Distance(transform.position, _bossTransform.position);
            
            if (distance > 2f)
            {
                Vector3 direction = (_bossTransform.position - transform.position).normalized;
                transform.position += direction * Speed * deltaTime;
            }
            
            // 跟随时也会攻击附近的敌人
            TryAttackNearbyEnemy();
        }
        
        private void AttackMove(float deltaTime)
        {
            // 寻找最近的敌人
            FindNearestEnemy();
            
            if (_targetEnemy != null && !_targetEnemy.IsDead)
            {
                float distance = Vector3.Distance(transform.position, _targetEnemy.transform.position);
                
                if (distance > 1.5f)
                {
                    // 移向敌人
                    Vector3 direction = (_targetEnemy.transform.position - transform.position).normalized;
                    transform.position += direction * Speed * deltaTime;
                }
                else
                {
                    // 攻击敌人
                    TryAttack(_targetEnemy);
                }
            }
        }
        
        private void HoldPosition(float deltaTime)
        {
            // 原地攻击
            TryAttackNearbyEnemy();
        }
        
        private void TryAttackNearbyEnemy()
        {
            if (_attackCooldown > 0) return;
            
            FindNearestEnemy();
            
            if (_targetEnemy != null)
            {
                float distance = Vector3.Distance(transform.position, _targetEnemy.transform.position);
                if (distance < 2f)
                {
                    TryAttack(_targetEnemy);
                }
            }
        }
        
        private void TryAttack(EnemyEntity enemy)
        {
            if (_attackCooldown > 0 || enemy == null || enemy.IsDead) return;
            
            enemy.TakeDamage(Damage, this);
            _attackCooldown = ATTACK_INTERVAL;
            
            CYLog.Debug($"[Employee] {name} 攻击 {enemy.name}");
        }
        
        private void FindNearestEnemy()
        {
            var enemies = CY.Entity.Manager.GetEntities("Enemy");
            float minDistance = float.MaxValue;
            _targetEnemy = null;
            
            foreach (var entity in enemies)
            {
                if (entity is EnemyEntity enemy && !enemy.IsDead)
                {
                    float distance = Vector3.Distance(transform.position, enemy.transform.position);
                    if (distance < minDistance)
                    {
                        minDistance = distance;
                        _targetEnemy = enemy;
                    }
                }
            }
        }
        
        /// <summary>
        /// 设置指令
        /// </summary>
        public void SetCommand(CommandType command)
        {
            CurrentCommand = command;
            CYLog.Debug($"[Employee] {name} 指令: {command}");
        }
    }
}
```

## 📝 第四步：注册和使用实体

### 4.1 在游戏入口中注册实体

```csharp
// LiangXinGame.cs 添加
[SerializeField] private GameObject _enemyPrefab;
[SerializeField] private GameObject _employeePrefab;

private void InitializeGame()
{
    // ... 其他初始化
    
    // 注册实体类型
    RegisterEntities();
}

private void RegisterEntities()
{
    if (_enemyPrefab != null)
    {
        CY.Entity.Register("Enemy", _enemyPrefab, 20);  // 预加载 20 个敌人
    }
    
    if (_employeePrefab != null)
    {
        CY.Entity.Register("Employee", _employeePrefab, 10);  // 预加载 10 个员工
    }
}
```

```csharp
// 招募员工
void HireEmployee(int configId, Vector3 position)
{
    var config = ConfigManager.GetEmployee(configId);
    if (config == null) return;
    
    // 检查资金
    if (!LiangXinGame.Game.Resources.SpendGold(config.HireCost))
    {
        CYLog.Warning("资金不足，无法招募");
        return;
    }
    
    var data = new UnitData
    {
        ConfigId = configId,
        SpawnPosition = position,
        Level = 1
    };
    
    var employee = CY.Entity.Show<EmployeeEntity>("Employee", data);
    CYLog.Info($"[招募员工] {config.Name}");
}
```

## ✅ 本章小结

| API | 说明 |
|-----|------|
| `CY.Entity.Register(type, prefab, count)` | 注册实体类型 |
| `CY.Entity.Show<T>(type, data)` | 显示实体 |
| `CY.Entity.Hide(entity)` | 隐藏实体 |
| `CY.Entity.HideAll(type)` | 隐藏所有指定类型 |
| `CY.Entity.Count(type)` | 获取实体数量 |

**EntityBase 生命周期**：
- `OnEntityShow()` - 显示时初始化
- `OnEntityUpdate()` - 每帧更新
- `OnEntityHide()` - 隐藏时清理
