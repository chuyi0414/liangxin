## 📝 第一步：完善 Enemy 类

### 1.1 更新 Enemy 脚本

打开 `Assets/_Game/Scripts/Unit/Enemy.cs`，替换为：

```csharp
using UnityEngine;
using CYFramework.Infrastructure;
using LiangXin.Data;
using LiangXin.Events;

namespace LiangXin.Unit
{
    /// <summary>
    /// 敌人单位
    /// </summary>
    public class Enemy : Unit
    {
        public override UnitFaction Faction => UnitFaction.Enemy;
        
        // ==================== 配置 ====================
        
        [Header("敌人配置")]
        [SerializeField] private EnemyConfig _enemyConfig;
        
        /// <summary>
        /// 敌人配置
        /// </summary>
        public EnemyConfig EnemyConfig => _enemyConfig;
        
        // ==================== AI ====================
        
        [Header("AI 状态")]
        [SerializeField] private EnemyAIType _aiType = EnemyAIType.Basic;
        
        /// <summary>
        /// 攻击目标（优先级：老板 > 员工 > 基地）
        /// </summary>
        private Transform _primaryTarget;
        
        /// <summary>
        /// 基地位置
        /// </summary>
        private Vector3 _basePosition = Vector3.zero;
        
        /// <summary>
        /// 仇恨目标
        /// </summary>
        private Unit _aggroTarget;
        
        /// <summary>
        /// 巡逻路径点
        /// </summary>
        private Vector3[] _patrolPoints;
        private int _currentPatrolIndex;
        
        // ==================== 技能 ====================
        
        [Header("技能")]
        [SerializeField] private float _skillTimer;
        
        // ==================== 生命周期 ====================
        
        protected override void Start()
        {
            base.Start();
            
            if (_enemyConfig != null)
            {
                _config = _enemyConfig;
                _stats.InitFromConfig(_enemyConfig);
                _aiType = _enemyConfig.aiType;
            }
            
            // 寻找老板作为主要目标
            var boss = FindObjectOfType<Boss>();
            if (boss != null)
            {
                _primaryTarget = boss.transform;
            }
        }
        
        protected override void Update()
        {
            base.Update();
            
            if (!IsAlive) return;
            
            // 更新技能冷却
            if (_skillTimer > 0)
            {
                _skillTimer -= Time.deltaTime;
            }
        }
        
        // ==================== AI 状态机 ====================
        
        protected override void UpdateIdle()
        {
            // 根据 AI 类型决定行为
            switch (_aiType)
            {
                case EnemyAIType.Basic:
                    BasicAI();
                    break;
                    
                case EnemyAIType.Ranged:
                    RangedAI();
                    break;
                    
                case EnemyAIType.Flanker:
                    FlankerAI();
                    break;
                    
                case EnemyAIType.Support:
                    SupportAI();
                    break;
                    
                case EnemyAIType.Suicide:
                    SuicideAI();
                    break;
                    
                case EnemyAIType.Stationary:
                    StationaryAI();
                    break;
                    
                case EnemyAIType.Patrol:
                    PatrolAI();
                    break;
                    
                default:
                    BasicAI();
                    break;
            }
        }
        
        protected override void UpdateAttack()
        {
            // 检查目标
            if (CurrentTarget == null || !CurrentTarget.IsAlive)
            {
                CurrentTarget = null;
                ChangeState(UnitState.Idle);
                return;
            }
            
            // 检查距离
            float distance = GetDistanceTo(CurrentTarget);
            if (distance > _stats.AttackRange)
            {
                // 追击
                MoveTo(CurrentTarget.transform.position);
                return;
            }
            
            // 攻击
            TryAttack();
            
            // 尝试使用技能
            if (_enemyConfig != null && _enemyConfig.hasSkill)
            {
                TryUseSkill();
            }
        }
        
        // ==================== AI 类型实现 ====================
        
        /// <summary>
        /// 基础 AI：直线冲向目标
        /// </summary>
        private void BasicAI()
        {
            // 寻找最近的敌方目标（老板或员工）
            Unit target = FindNearestPlayerUnit();
            
            if (target != null)
            {
                float distance = GetDistanceTo(target);
                
                if (distance <= _stats.AttackRange)
                {
                    CurrentTarget = target;
                    ChangeState(UnitState.Attack);
                }
                else if (distance <= (_enemyConfig?.aggroRange ?? 15f))
                {
                    // 进入仇恨范围，追击
                    MoveTo(target.transform.position);
                }
                else
                {
                    // 向基地移动
                    MoveTo(_basePosition);
                }
            }
            else
            {
                // 没有目标，向基地移动
                MoveTo(_basePosition);
            }
        }
        
        /// <summary>
        /// 远程 AI：保持距离攻击
        /// </summary>
        private void RangedAI()
        {
            Unit target = FindNearestPlayerUnit();
            
            if (target != null)
            {
                float distance = GetDistanceTo(target);
                float preferredRange = _stats.AttackRange * 0.8f;
                
                if (distance < preferredRange * 0.5f)
                {
                    // 太近了，后退
                    Vector3 direction = (transform.position - target.transform.position).normalized;
                    MoveTo(transform.position + direction * 3f);
                }
                else if (distance <= _stats.AttackRange)
                {
                    // 在攻击范围内，攻击
                    CurrentTarget = target;
                    ChangeState(UnitState.Attack);
                }
                else if (distance <= (_enemyConfig?.aggroRange ?? 15f))
                {
                    // 靠近到攻击范围
                    Vector3 direction = (target.transform.position - transform.position).normalized;
                    Vector3 targetPos = target.transform.position - direction * preferredRange;
                    MoveTo(targetPos);
                }
                else
                {
                    MoveTo(_basePosition);
                }
            }
            else
            {
                MoveTo(_basePosition);
            }
        }
        
        /// <summary>
        /// 侧翼 AI：绕后包抄
        /// </summary>
        private void FlankerAI()
        {
            Unit target = FindNearestPlayerUnit();
            
            if (target != null)
            {
                float distance = GetDistanceTo(target);
                
                if (distance <= _stats.AttackRange)
                {
                    CurrentTarget = target;
                    ChangeState(UnitState.Attack);
                }
                else
                {
                    // 绕到目标侧面或后面
                    Vector3 toTarget = target.transform.position - transform.position;
                    Vector3 perpendicular = new Vector3(-toTarget.y, toTarget.x, 0).normalized;
                    
                    // 随机选择左侧或右侧
                    if (UnitId % 2 == 0)
                        perpendicular = -perpendicular;
                    
                    Vector3 flankPosition = target.transform.position + perpendicular * 3f;
                    MoveTo(flankPosition);
                }
            }
            else
            {
                MoveTo(_basePosition);
            }
        }
        
        /// <summary>
        /// 辅助 AI：给其他敌人加 Buff
        /// </summary>
        private void SupportAI()
        {
            // 寻找血量最低的友方敌人
            Enemy allyToHeal = FindLowestHPAlly();
            
            if (allyToHeal != null && allyToHeal != this)
            {
                float distance = GetDistanceTo(allyToHeal);
                
                if (distance <= _stats.AttackRange)
                {
                    // 治疗友军
                    if (_skillTimer <= 0)
                    {
                        allyToHeal.Heal(20);
                        _skillTimer = _enemyConfig?.skillCooldown ?? 5f;
                        CYLog.Debug($"[{_enemyConfig?.unitName}] 治疗了 {allyToHeal.EnemyConfig?.unitName}");
                    }
                }
                else
                {
                    MoveTo(allyToHeal.transform.position);
                }
            }
            else
            {
                // 没有需要治疗的，正常攻击
                BasicAI();
            }
        }
        
        /// <summary>
        /// 自爆 AI：冲向目标自爆
        /// </summary>
        private void SuicideAI()
        {
            Unit target = FindNearestPlayerUnit();
            
            if (target != null)
            {
                float distance = GetDistanceTo(target);
                
                if (distance <= 1.5f)
                {
                    // 自爆！
                    Explode();
                }
                else
                {
                    // 全速冲向目标
                    MoveTo(target.transform.position);
                }
            }
            else
            {
                MoveTo(_basePosition);
            }
        }
        
        /// <summary>
        /// 固定 AI：原地不动
        /// </summary>
        private void StationaryAI()
        {
            // 只攻击进入范围的敌人
            Unit target = FindNearestPlayerUnit();
            
            if (target != null)
            {
                float distance = GetDistanceTo(target);
                
                if (distance <= _stats.AttackRange)
                {
                    CurrentTarget = target;
                    ChangeState(UnitState.Attack);
                }
            }
        }
        
        /// <summary>
        /// 巡逻 AI：按路径移动
        /// </summary>
        private void PatrolAI()
        {
            // 简化：没有预设路径就用基础 AI
            if (_patrolPoints == null || _patrolPoints.Length == 0)
            {
                BasicAI();
                return;
            }
            
            // 检查是否有敌人进入仇恨范围
            Unit target = FindNearestPlayerUnit();
            if (target != null)
            {
                float distance = GetDistanceTo(target);
                if (distance <= (_enemyConfig?.aggroRange ?? 10f))
                {
                    CurrentTarget = target;
                    ChangeState(UnitState.Attack);
                    return;
                }
            }
            
            // 巡逻
            Vector3 patrolTarget = _patrolPoints[_currentPatrolIndex];
            float distToPatrol = Vector3.Distance(transform.position, patrolTarget);
            
            if (distToPatrol < 0.5f)
            {
                _currentPatrolIndex = (_currentPatrolIndex + 1) % _patrolPoints.Length;
            }
            else
            {
                MoveTo(patrolTarget);
            }
        }
        
        // ==================== 辅助方法 ====================
        
        /// <summary>
        /// 寻找最近的玩家单位
        /// </summary>
        private Unit FindNearestPlayerUnit()
        {
            float searchRange = _enemyConfig?.aggroRange ?? 15f;
            Collider2D[] hits = Physics2D.OverlapCircleAll(transform.position, searchRange);
            
            Unit nearest = null;
            float nearestDist = float.MaxValue;
            
            foreach (var hit in hits)
            {
                Unit unit = hit.GetComponent<Unit>();
                if (unit != null && unit.Faction == UnitFaction.Player && unit.IsAlive)
                {
                    float dist = GetDistanceTo(unit);
                    
                    // 老板优先级更高
                    if (unit is Boss)
                    {
                        dist *= 0.5f;
                    }
                    
                    if (dist < nearestDist)
                    {
                        nearestDist = dist;
                        nearest = unit;
                    }
                }
            }
            
            return nearest;
        }
        
        /// <summary>
        /// 寻找血量最低的友方敌人
        /// </summary>
        private Enemy FindLowestHPAlly()
        {
            var enemies = FindObjectsOfType<Enemy>();
            Enemy lowest = null;
            float lowestPercent = 1f;
            
            foreach (var enemy in enemies)
            {
                if (enemy != this && enemy.IsAlive)
                {
                    float percent = enemy.Stats.HPPercent;
                    if (percent < lowestPercent && percent < 0.8f)
                    {
                        lowestPercent = percent;
                        lowest = enemy;
                    }
                }
            }
            
            return lowest;
        }
        
        /// <summary>
        /// 自爆
        /// </summary>
        private void Explode()
        {
            CYLog.Info($"[{_enemyConfig?.unitName}] 自爆！");
            
            // 对范围内所有玩家单位造成伤害
            float explosionRadius = 2f;
            Collider2D[] hits = Physics2D.OverlapCircleAll(transform.position, explosionRadius);
            
            foreach (var hit in hits)
            {
                Unit unit = hit.GetComponent<Unit>();
                if (unit != null && unit.Faction == UnitFaction.Player && unit.IsAlive)
                {
                    unit.TakeDamage(_stats.Attack * 2, this);
                }
            }
            
            // 自己死亡
            Die(null);
        }
        
        // ==================== 技能 ====================
        
        private void TryUseSkill()
        {
            if (_skillTimer > 0) return;
            
            // 简化：技能就是造成额外伤害
            if (CurrentTarget != null)
            {
                CurrentTarget.TakeDamage(_stats.Attack, this);
                _skillTimer = _enemyConfig?.skillCooldown ?? 5f;
            }
        }
        
        // ==================== 死亡 ====================
        
        public override void Die(Unit killer = null)
        {
            if (CurrentState == UnitState.Dead) return;
            
            // 掉落奖励
            if (_enemyConfig != null)
            {
                GameManager.Instance.AddGold(_enemyConfig.goldReward);
            }
            
            base.Die(killer);
        }
        
        // ==================== 设置 ====================
        
        /// <summary>
        /// 设置敌人配置
        /// </summary>
        public void SetEnemyConfig(EnemyConfig config)
        {
            _enemyConfig = config;
            _config = config;
            if (config != null)
            {
                _stats.InitFromConfig(config);
                _aiType = config.aiType;
            }
        }
        
        /// <summary>
        /// 设置巡逻路径
        /// </summary>
        public void SetPatrolPoints(Vector3[] points)
        {
            _patrolPoints = points;
            _currentPatrolIndex = 0;
        }
    }
}
```

## 📝 第二步：创建波次管理器

### 2.1 创建 WaveManager 脚本

1. 右键 `Assets/_Game/Scripts/Battle`
2. `Create > C# Script`
3. 命名为 `WaveManager`

### 2.2 编写代码

```csharp
using UnityEngine;
using CYFramework.Infrastructure;
using CYFramework.Core.Event;
using LiangXin.Events;
using LiangXin.Data;
using LiangXin.Unit;
using System.Collections;
using System.Collections.Generic;

namespace LiangXin.Battle
{
    /// <summary>
    /// 波次管理器
    /// 负责生成敌人和管理波次
    /// </summary>
    public class WaveManager : MonoBehaviour
    {
        // ==================== 单例 ====================
        
        public static WaveManager Instance { get; private set; }
        
        // ==================== 配置 ====================
        
        [Header("生成配置")]
        [Tooltip("敌人预制体")]
        [SerializeField] private GameObject _enemyPrefab;
        
        [Tooltip("生成点")]
        [SerializeField] private Transform[] _spawnPoints;
        
        [Tooltip("波次间隔")]
        [SerializeField] private float _waveInterval = 10f;
        
        [Tooltip("准备时间")]
        [SerializeField] private float _prepareTime = 10f;
        
        [Header("难度曲线")]
        [Tooltip("基础敌人数量")]
        [SerializeField] private int _baseEnemyCount = 5;
        
        [Tooltip("每波增加的敌人")]
        [SerializeField] private int _enemiesPerWave = 2;
        
        [Tooltip("精英出现的波次")]
        [SerializeField] private int _eliteStartWave = 5;
        
        [Tooltip("BOSS 出现的波次")]
        [SerializeField] private int _bossStartWave = 10;
        
        // ==================== 状态 ====================
        
        [Header("状态")]
        [SerializeField] private int _currentWave = 0;
        [SerializeField] private int _enemiesAlive = 0;
        [SerializeField] private int _enemiesKilledThisWave = 0;
        [SerializeField] private bool _isWaveActive = false;
        
        /// <summary>
        /// 当前波次
        /// </summary>
        public int CurrentWave => _currentWave;
        
        /// <summary>
        /// 存活敌人数
        /// </summary>
        public int EnemiesAlive => _enemiesAlive;
        
        /// <summary>
        /// 波次是否进行中
        /// </summary>
        public bool IsWaveActive => _isWaveActive;
        
        // ==================== 服务 ====================
        
        private EventBus _eventBus;
        
        // ==================== 生命周期 ====================
        
        void Awake()
        {
            if (Instance != null && Instance != this)
            {
                Destroy(gameObject);
                return;
            }
            Instance = this;
        }
        
        void Start()
        {
            _eventBus = ServiceLocator.Get<EventBus>();
            
            // 订阅事件
            _eventBus.Subscribe<GameStartEvent>(OnGameStart, this);
            _eventBus.Subscribe<UnitDiedEvent>(OnUnitDied, this);
            
            // 如果没有生成点，创建默认的
            if (_spawnPoints == null || _spawnPoints.Length == 0)
            {
                CreateDefaultSpawnPoints();
            }
        }
        
        void OnDestroy()
        {
            _eventBus?.UnsubscribeAll(this);
        }
        
        // ==================== 默认生成点 ====================
        
        private void CreateDefaultSpawnPoints()
        {
            // 创建四个方向的生成点
            var parent = new GameObject("SpawnPoints");
            _spawnPoints = new Transform[4];
            
            Vector3[] positions = new Vector3[]
            {
                new Vector3(0, 10, 0),   // 北
                new Vector3(10, 0, 0),   // 东
                new Vector3(0, -10, 0),  // 南
                new Vector3(-10, 0, 0)   // 西
            };
            
            for (int i = 0; i < 4; i++)
            {
                var point = new GameObject($"SpawnPoint_{i}");
                point.transform.parent = parent.transform;
                point.transform.position = positions[i];
                _spawnPoints[i] = point.transform;
            }
            
            CYLog.Info("[WaveManager] 创建了默认生成点");
        }
        
        // ==================== 事件处理 ====================
        
        private void OnGameStart(ref GameStartEvent evt)
        {
            _currentWave = 0;
            _enemiesAlive = 0;
            StartNextWave();
        }
        
        private void OnUnitDied(ref UnitDiedEvent evt)
        {
            if (evt.IsEnemy)
            {
                _enemiesAlive--;
                _enemiesKilledThisWave++;
                
                CYLog.Debug($"[WaveManager] 敌人死亡，剩余: {_enemiesAlive}");
                
                // 检查波次是否结束
                if (_enemiesAlive <= 0 && _isWaveActive)
                {
                    EndCurrentWave();
                }
            }
        }
        
        // ==================== 波次控制 ====================
        
        /// <summary>
        /// 开始下一波
        /// </summary>
        public void StartNextWave()
        {
            _currentWave++;
            _enemiesKilledThisWave = 0;
            
            CYLog.Info($"[WaveManager] ===== 波次 {_currentWave} 开始 =====");
            
            // 发布事件
            var evt = new WaveStartEvent
            {
                WaveNumber = _currentWave,
                PrepareTime = _prepareTime
            };
            _eventBus.Post(ref evt);
            
            // 开始生成敌人
            StartCoroutine(SpawnWaveCoroutine());
        }
        
        /// <summary>
        /// 结束当前波次
        /// </summary>
        private void EndCurrentWave()
        {
            _isWaveActive = false;
            
            CYLog.Info($"[WaveManager] ===== 波次 {_currentWave} 结束，击杀: {_enemiesKilledThisWave} =====");
            
            // 发布事件
            var evt = new WaveEndEvent
            {
                WaveNumber = _currentWave,
                EnemiesKilled = _enemiesKilledThisWave
            };
            _eventBus.Post(ref evt);
            
            // 延迟后开始下一波
            StartCoroutine(StartNextWaveAfterDelay());
        }
        
        private IEnumerator StartNextWaveAfterDelay()
        {
            yield return new WaitForSeconds(_waveInterval);
            
            if (GameManager.Instance.CurrentState == GameState.Battle ||
                GameManager.Instance.CurrentState == GameState.Prepare)
            {
                StartNextWave();
            }
        }
        
        // ==================== 敌人生成 ====================
        
        private IEnumerator SpawnWaveCoroutine()
        {
            // 准备阶段
            GameManager.Instance.ChangeState(GameState.Prepare);
            
            // 显示预警信息
            ShowWaveWarning();
            
            yield return new WaitForSeconds(_prepareTime);
            
            // 开始战斗
            GameManager.Instance.ChangeState(GameState.Battle);
            _isWaveActive = true;
            
            // 计算本波敌人数量
            int enemyCount = CalculateEnemyCount();
            
            // 生成敌人
            for (int i = 0; i < enemyCount; i++)
            {
                SpawnEnemy();
                yield return new WaitForSeconds(0.5f); // 间隔生成
            }
            
            CYLog.Info($"[WaveManager] 生成了 {enemyCount} 个敌人");
        }
        
        /// <summary>
        /// 显示波次预警
        /// </summary>
        private void ShowWaveWarning()
        {
            // 随机选择进攻方向
            List<string> directions = new List<string>();
            
            int activeSpawnPoints = Mathf.Min(_currentWave, _spawnPoints.Length);
            for (int i = 0; i < activeSpawnPoints; i++)
            {
                string[] dirNames = { "北方", "东方", "南方", "西方" };
                if (i < dirNames.Length)
                {
                    directions.Add(dirNames[i]);
                }
            }
            
            string warning = $"[波次 {_currentWave}] 敌人将从 {string.Join("、", directions)} 进攻！";
            CYLog.Warning(warning);
        }
        
        /// <summary>
        /// 计算本波敌人数量
        /// </summary>
        private int CalculateEnemyCount()
        {
            return _baseEnemyCount + (_currentWave - 1) * _enemiesPerWave;
        }
        
        /// <summary>
        /// 生成一个敌人
        /// </summary>
        private void SpawnEnemy()
        {
            if (_enemyPrefab == null)
            {
                CYLog.Error("[WaveManager] 敌人预制体未设置！");
                return;
            }
            
            // 随机选择生成点
            int spawnIndex = Random.Range(0, Mathf.Min(_currentWave, _spawnPoints.Length));
            Vector3 spawnPos = _spawnPoints[spawnIndex].position;
            
            // 添加随机偏移
            spawnPos += new Vector3(
                Random.Range(-2f, 2f),
                Random.Range(-2f, 2f),
                0
            );
            
            // 生成敌人
            GameObject enemyObj = Instantiate(_enemyPrefab, spawnPos, Quaternion.identity);
            Enemy enemy = enemyObj.GetComponent<Enemy>();
            
            if (enemy != null)
            {
                _enemiesAlive++;
            }
        }
        
        /// <summary>
        /// 生成指定类型的敌人
        /// </summary>
        public void SpawnEnemy(EnemyConfig config, Vector3 position)
        {
            if (_enemyPrefab == null) return;
            
            GameObject enemyObj = Instantiate(_enemyPrefab, position, Quaternion.identity);
            Enemy enemy = enemyObj.GetComponent<Enemy>();
            
            if (enemy != null)
            {
                enemy.SetEnemyConfig(config);
                _enemiesAlive++;
            }
        }
    }
}
```

## 📝 第四步：测试波次系统
