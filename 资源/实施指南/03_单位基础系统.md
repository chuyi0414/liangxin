# 03. 单位基础系统

## 🎯 本章目标

1. 创建 Unit 基类（所有单位的父类）
2. 实现单位属性系统
3. 实现单位状态机
4. 实现生命周期管理

完成后：
- 所有单位有统一的属性和行为
- 单位可以移动、攻击、受伤、死亡

## 📖 概念讲解：单位系统设计

```
┌─────────────────────────────────────────────────────┐
│                     Unit (基类)                      │
│  - 属性：HP, Attack, Defense, MoveSpeed...          │
│  - 状态：Idle, Move, Attack, Dead, Stunned...       │
│  - 方法：TakeDamage(), Die(), Move()...             │
└─────────────────────────────────────────────────────┘
              ↑                          ↑
              │                          │
   ┌──────────┴──────────┐    ┌──────────┴──────────┐
   │     Employee        │    │       Enemy         │
   │   (友方员工)         │    │     (敌方单位)       │
   │ - 等级、词条、技能   │    │ - AI行为、掉落       │
   └─────────────────────┘    └─────────────────────┘
              ↑
              │
   ┌──────────┴──────────┐
   │       Boss          │
   │   (良心老板/玩家)    │
   │ - 指挥系统、特殊能力 │
   └─────────────────────┘
```

## 📝 第一步：创建单位属性类

### 1.1 创建 UnitStats 脚本

1. 右键 `Assets/_Game/Scripts/Unit`
2. `Create > C# Script`
3. 命名为 `UnitStats`

### 1.2 编写代码

```csharp
using UnityEngine;
using System;

namespace LiangXin.Unit
{
    /// <summary>
    /// 单位属性
    /// 管理单位的所有数值属性
    /// </summary>
    [System.Serializable]
    public class UnitStats
    {
        // ==================== 基础属性 ====================
        
        [Header("生命值")]
        [SerializeField] private int _maxHP = 100;
        [SerializeField] private int _currentHP = 100;
        
        [Header("战斗属性")]
        [SerializeField] private int _attack = 10;
        [SerializeField] private int _defense = 0;
        [SerializeField] private float _attackSpeed = 1f;
        [SerializeField] private float _attackRange = 2f;
        
        [Header("移动属性")]
        [SerializeField] private float _moveSpeed = 5f;
        
        [Header("暴击属性")]
        [SerializeField] private float _critRate = 0f;      // 暴击率 0-1
        [SerializeField] private float _critDamage = 1.5f;  // 暴击伤害倍率
        
        // ==================== 属性访问器 ====================
        
        public int MaxHP
        {
            get => _maxHP;
            set => _maxHP = Mathf.Max(1, value);
        }
        
        public int CurrentHP
        {
            get => _currentHP;
            set
            {
                int oldHP = _currentHP;
                _currentHP = Mathf.Clamp(value, 0, _maxHP);
                if (_currentHP != oldHP)
                {
                    OnHPChanged?.Invoke(_currentHP, _maxHP);
                }
            }
        }
        
        public int Attack
        {
            get => _attack;
            set => _attack = Mathf.Max(0, value);
        }
        
        public int Defense
        {
            get => _defense;
            set => _defense = Mathf.Max(0, value);
        }
        
        public float AttackSpeed
        {
            get => _attackSpeed;
            set => _attackSpeed = Mathf.Max(0.1f, value);
        }
        
        public float AttackRange
        {
            get => _attackRange;
            set => _attackRange = Mathf.Max(0.5f, value);
        }
        
        public float MoveSpeed
        {
            get => _moveSpeed;
            set => _moveSpeed = Mathf.Max(0f, value);
        }
        
        public float CritRate
        {
            get => _critRate;
            set => _critRate = Mathf.Clamp01(value);
        }
        
        public float CritDamage
        {
            get => _critDamage;
            set => _critDamage = Mathf.Max(1f, value);
        }
        
        // ==================== 计算属性 ====================
        
        /// <summary>
        /// 血量百分比 (0-1)
        /// </summary>
        public float HPPercent => _maxHP > 0 ? (float)_currentHP / _maxHP : 0f;
        
        /// <summary>
        /// 是否存活
        /// </summary>
        public bool IsAlive => _currentHP > 0;
        
        /// <summary>
        /// 攻击间隔（秒）
        /// </summary>
        public float AttackInterval => _attackSpeed > 0 ? 1f / _attackSpeed : 9999f;
        
        // ==================== 事件 ====================
        
        /// <summary>
        /// 血量变化事件 (当前HP, 最大HP)
        /// </summary>
        public event Action<int, int> OnHPChanged;
        
        // ==================== 方法 ====================
        
        /// <summary>
        /// 从配置初始化
        /// </summary>
        public void InitFromConfig(Data.UnitConfig config)
        {
            _maxHP = config.maxHP;
            _currentHP = config.maxHP;
            _attack = config.attack;
            _defense = config.defense;
            _attackSpeed = config.attackSpeed;
            _attackRange = config.attackRange;
            _moveSpeed = config.moveSpeed;
        }
        
        /// <summary>
        /// 复制属性
        /// </summary>
        public void CopyFrom(UnitStats other)
        {
            _maxHP = other._maxHP;
            _currentHP = other._currentHP;
            _attack = other._attack;
            _defense = other._defense;
            _attackSpeed = other._attackSpeed;
            _attackRange = other._attackRange;
            _moveSpeed = other._moveSpeed;
            _critRate = other._critRate;
            _critDamage = other._critDamage;
        }
        
        /// <summary>
        /// 回满血
        /// </summary>
        public void FullHeal()
        {
            CurrentHP = _maxHP;
        }
        
        /// <summary>
        /// 计算最终伤害（考虑暴击）
        /// </summary>
        public int CalculateDamage(out bool isCrit)
        {
            isCrit = UnityEngine.Random.value < _critRate;
            int damage = _attack;
            if (isCrit)
            {
                damage = Mathf.RoundToInt(damage * _critDamage);
            }
            return damage;
        }
    }
}
```

## 📝 第三步：创建 Unit 基类

### 3.1 创建 Unit 脚本

1. 右键 `Assets/_Game/Scripts/Unit`
2. `Create > C# Script`
3. 命名为 `Unit`

### 3.2 编写代码

```csharp
using UnityEngine;
using CYFramework.Infrastructure;
using CYFramework.Core.Event;
using LiangXin.Data;
using LiangXin.Events;

namespace LiangXin.Unit
{
    /// <summary>
    /// 单位基类
    /// 所有友方和敌方单位的父类
    /// </summary>
    public abstract class Unit : MonoBehaviour
    {
        // ==================== 配置 ====================
        
        [Header("单位配置")]
        [SerializeField] protected UnitConfig _config;
        
        // ==================== 属性 ====================
        
        [Header("运行时属性")]
        [SerializeField] protected UnitStats _stats = new UnitStats();
        
        /// <summary>
        /// 单位属性
        /// </summary>
        public UnitStats Stats => _stats;
        
        /// <summary>
        /// 单位配置
        /// </summary>
        public UnitConfig Config => _config;
        
        // ==================== 状态 ====================
        
        /// <summary>
        /// 当前状态
        /// </summary>
        public UnitState CurrentState { get; protected set; } = UnitState.None;
        
        /// <summary>
        /// 阵营
        /// </summary>
        public abstract UnitFaction Faction { get; }
        
        /// <summary>
        /// 唯一 ID（运行时分配）
        /// </summary>
        public int UnitId { get; private set; }
        
        /// <summary>
        /// 是否存活
        /// </summary>
        public bool IsAlive => _stats.IsAlive && CurrentState != UnitState.Dead;
        
        // ==================== 组件引用 ====================
        
        protected Rigidbody2D _rigidbody;
        protected SpriteRenderer _spriteRenderer;
        protected Animator _animator;
        
        // ==================== 目标与移动 ====================
        
        /// <summary>
        /// 当前目标
        /// </summary>
        public Unit CurrentTarget { get; protected set; }
        
        /// <summary>
        /// 移动目标位置
        /// </summary>
        protected Vector3 _moveTarget;
        
        /// <summary>
        /// 是否有移动目标
        /// </summary>
        protected bool _hasMoveTarget;
        
        // ==================== 攻击 ====================
        
        /// <summary>
        /// 攻击冷却计时器
        /// </summary>
        protected float _attackCooldown;
        
        // ==================== 服务引用 ====================
        
        protected EventBus _eventBus;
        
        // ==================== 静态 ID 计数器 ====================
        
        private static int _nextUnitId = 1;
        
        // ==================== 生命周期 ====================
        
        protected virtual void Awake()
        {
            // 分配唯一 ID
            UnitId = _nextUnitId++;
            
            // 获取组件
            _rigidbody = GetComponent<Rigidbody2D>();
            _spriteRenderer = GetComponent<SpriteRenderer>();
            _animator = GetComponent<Animator>();
        }
        
        protected virtual void Start()
        {
            // 获取事件总线
            _eventBus = ServiceLocator.Get<EventBus>();
            
            // 初始化属性
            if (_config != null)
            {
                _stats.InitFromConfig(_config);
            }
            
            // 进入空闲状态
            ChangeState(UnitState.Idle);
            
            // 发布单位生成事件
            var evt = new UnitSpawnedEvent
            {
                UnitId = UnitId,
                IsEnemy = Faction == UnitFaction.Enemy,
                Position = transform.position
            };
            _eventBus.Post(ref evt);
        }
        
        protected virtual void Update()
        {
            if (!IsAlive) return;
            
            // 更新攻击冷却
            if (_attackCooldown > 0)
            {
                _attackCooldown -= Time.deltaTime;
            }
            
            // 更新状态机
            UpdateStateMachine();
        }
        
        protected virtual void FixedUpdate()
        {
            if (!IsAlive) return;
            
            // 处理移动
            if (CurrentState == UnitState.Move && _hasMoveTarget)
            {
                ProcessMovement();
            }
        }
        
        // ==================== 状态机 ====================
        
        /// <summary>
        /// 切换状态
        /// </summary>
        public virtual void ChangeState(UnitState newState)
        {
            if (CurrentState == newState) return;
            
            // 退出旧状态
            OnStateExit(CurrentState);
            
            var oldState = CurrentState;
            CurrentState = newState;
            
            // 进入新状态
            OnStateEnter(newState);
        }
        
        /// <summary>
        /// 退出状态
        /// </summary>
        protected virtual void OnStateExit(UnitState state)
        {
            // 子类可重写
        }
        
        /// <summary>
        /// 进入状态
        /// </summary>
        protected virtual void OnStateEnter(UnitState state)
        {
            // 播放动画
            switch (state)
            {
                case UnitState.Idle:
                    PlayAnimation("Idle");
                    break;
                case UnitState.Move:
                    PlayAnimation("Move");
                    break;
                case UnitState.Attack:
                    PlayAnimation("Attack");
                    break;
                case UnitState.Dead:
                    PlayAnimation("Dead");
                    break;
            }
        }
        
        /// <summary>
        /// 更新状态机
        /// </summary>
        protected virtual void UpdateStateMachine()
        {
            switch (CurrentState)
            {
                case UnitState.Idle:
                    UpdateIdle();
                    break;
                case UnitState.Move:
                    UpdateMove();
                    break;
                case UnitState.Attack:
                    UpdateAttack();
                    break;
            }
        }
        
        protected virtual void UpdateIdle()
        {
            // 子类实现
        }
        
        protected virtual void UpdateMove()
        {
            // 检查是否到达目标
            if (_hasMoveTarget)
            {
                float distance = Vector3.Distance(transform.position, _moveTarget);
                if (distance < 0.1f)
                {
                    _hasMoveTarget = false;
                    ChangeState(UnitState.Idle);
                }
            }
        }
        
        protected virtual void UpdateAttack()
        {
            // 子类实现
        }
        
        // ==================== 移动 ====================
        
        /// <summary>
        /// 移动到目标位置
        /// </summary>
        public virtual void MoveTo(Vector3 position)
        {
            _moveTarget = position;
            _hasMoveTarget = true;
            ChangeState(UnitState.Move);
        }
        
        /// <summary>
        /// 停止移动
        /// </summary>
        public virtual void StopMove()
        {
            _hasMoveTarget = false;
            if (_rigidbody != null)
            {
                _rigidbody.velocity = Vector2.zero;
            }
            ChangeState(UnitState.Idle);
        }
        
        /// <summary>
        /// 处理移动（在 FixedUpdate 中调用）
        /// </summary>
        protected virtual void ProcessMovement()
        {
            if (_rigidbody == null) return;
            
            Vector3 direction = (_moveTarget - transform.position).normalized;
            _rigidbody.velocity = direction * _stats.MoveSpeed;
            
            // 翻转精灵
            if (_spriteRenderer != null && direction.x != 0)
            {
                _spriteRenderer.flipX = direction.x < 0;
            }
        }
        
        // ==================== 战斗 ====================
        
        /// <summary>
        /// 设置攻击目标
        /// </summary>
        public virtual void SetTarget(Unit target)
        {
            CurrentTarget = target;
        }
        
        /// <summary>
        /// 尝试攻击
        /// </summary>
        public virtual bool TryAttack()
        {
            if (!CanAttack()) return false;
            
            PerformAttack();
            _attackCooldown = _stats.AttackInterval;
            return true;
        }
        
        /// <summary>
        /// 是否可以攻击
        /// </summary>
        protected virtual bool CanAttack()
        {
            if (!IsAlive) return false;
            if (_attackCooldown > 0) return false;
            if (CurrentTarget == null || !CurrentTarget.IsAlive) return false;
            
            // 检查距离
            float distance = Vector3.Distance(transform.position, CurrentTarget.transform.position);
            if (distance > _stats.AttackRange) return false;
            
            return true;
        }
        
        /// <summary>
        /// 执行攻击
        /// </summary>
        protected virtual void PerformAttack()
        {
            if (CurrentTarget == null) return;
            
            // 计算伤害
            int damage = _stats.CalculateDamage(out bool isCrit);
            
            // 对目标造成伤害
            CurrentTarget.TakeDamage(damage, this, isCrit);
            
            CYLog.Debug($"[{_config?.unitName}] 攻击 [{CurrentTarget.Config?.unitName}] 造成 {damage} 伤害{(isCrit ? " (暴击!)" : "")}");
        }
        
        /// <summary>
        /// 受到伤害
        /// </summary>
        public virtual void TakeDamage(int damage, Unit attacker, bool isCrit = false)
        {
            if (!IsAlive) return;
            
            // 计算实际伤害（减去防御）
            int actualDamage = Mathf.Max(1, damage - _stats.Defense);
            
            // 扣血
            _stats.CurrentHP -= actualDamage;
            
            CYLog.Debug($"[{_config?.unitName}] 受到 {actualDamage} 伤害，剩余 HP: {_stats.CurrentHP}/{_stats.MaxHP}");
            
            // 检查死亡
            if (!_stats.IsAlive)
            {
                Die(attacker);
            }
        }
        
        /// <summary>
        /// 治疗
        /// </summary>
        public virtual void Heal(int amount, Unit healer = null)
        {
            if (!IsAlive) return;
            
            int oldHP = _stats.CurrentHP;
            _stats.CurrentHP += amount;
            int actualHeal = _stats.CurrentHP - oldHP;
            
            CYLog.Debug($"[{_config?.unitName}] 恢复 {actualHeal} HP，当前: {_stats.CurrentHP}/{_stats.MaxHP}");
        }
        
        /// <summary>
        /// 死亡
        /// </summary>
        public virtual void Die(Unit killer = null)
        {
            if (CurrentState == UnitState.Dead) return;
            
            ChangeState(UnitState.Dead);
            
            // 发布死亡事件
            var evt = new UnitDiedEvent
            {
                UnitId = UnitId,
                IsEnemy = Faction == UnitFaction.Enemy,
                KillerId = killer?.UnitId ?? 0,
                Position = transform.position
            };
            _eventBus.Post(ref evt);
            
            CYLog.Info($"[{_config?.unitName}] 死亡");
            
            // 延迟销毁
            Destroy(gameObject, 2f);
        }
        
        // ==================== 辅助方法 ====================
        
        /// <summary>
        /// 播放动画
        /// </summary>
        protected virtual void PlayAnimation(string animName)
        {
            if (_animator != null)
            {
                _animator.Play(animName);
            }
        }
        
        /// <summary>
        /// 设置配置
        /// </summary>
        public virtual void SetConfig(UnitConfig config)
        {
            _config = config;
            if (config != null)
            {
                _stats.InitFromConfig(config);
            }
        }
        
        /// <summary>
        /// 获取到目标的距离
        /// </summary>
        public float GetDistanceTo(Unit other)
        {
            if (other == null) return float.MaxValue;
            return Vector3.Distance(transform.position, other.transform.position);
        }
        
        /// <summary>
        /// 获取到位置的距离
        /// </summary>
        public float GetDistanceTo(Vector3 position)
        {
            return Vector3.Distance(transform.position, position);
        }
        
        // ==================== Gizmos（调试用）====================
        
        #if UNITY_EDITOR
        protected virtual void OnDrawGizmosSelected()
        {
            // 绘制攻击范围
            Gizmos.color = Color.red;
            Gizmos.DrawWireSphere(transform.position, _stats?.AttackRange ?? 2f);
        }
        #endif
    }
}
```
