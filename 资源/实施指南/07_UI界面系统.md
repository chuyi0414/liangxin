## 📝 第一步：创建 HUD 面板

### 1.1 创建 GameHUDPanel 脚本

1. 右键 `Assets/_Game/Scripts/UI`
2. `Create > C# Script`
3. 命名为 `GameHUDPanel`

### 1.2 编写代码

```csharp
using UnityEngine;
using UnityEngine.UI;
using CYFramework.Infrastructure;
using CYFramework.Core.Event;
using CYFramework.Modules.UI;
using LiangXin.Events;
using LiangXin.Unit;

namespace LiangXin.UI
{
    /// <summary>
    /// 游戏 HUD 面板
    /// 显示资源、波次、血条等信息
    /// </summary>
    [UIPrefab("UI/Panels/GameHUDPanel")]
    public class GameHUDPanel : UIPanel
    {
        // ==================== UI 引用 ====================
        
        [Header("资源显示")]
        [SerializeField] private Text _goldText;
        [SerializeField] private Text _conscienceText;
        [SerializeField] private Text _waveText;
        
        [Header("老板血条")]
        [SerializeField] private Slider _bossHPBar;
        [SerializeField] private Text _bossHPText;
        
        [Header("选中员工信息")]
        [SerializeField] private GameObject _selectedPanel;
        [SerializeField] private Text _selectedNameText;
        [SerializeField] private Text _selectedLevelText;
        [SerializeField] private Slider _selectedHPBar;
        [SerializeField] private Button _upgradeButton;
        [SerializeField] private Text _upgradeCostText;
        
        [Header("状态提示")]
        [SerializeField] private Text _stateText;
        
        // ==================== 引用 ====================
        
        private EventBus _eventBus;
        private Boss _boss;
        private Employee _selectedEmployee;
        
        // ==================== 生命周期 ====================
        
        public override UILayer Layer => UILayer.Main;
        
        protected override void OnBindUI()
        {
            base.OnBindUI();
            
            if (_upgradeButton != null)
            {
                _upgradeButton.onClick.AddListener(OnUpgradeClicked);
            }
        }
        
        protected override void OnUnbindUI()
        {
            base.OnUnbindUI();
            
            if (_upgradeButton != null)
            {
                _upgradeButton.onClick.RemoveListener(OnUpgradeClicked);
            }
        }
        
        protected override void OnShow(object data)
        {
            // 获取服务
            _eventBus = ServiceLocator.Get<EventBus>();
            
            // 订阅事件
            _eventBus.Subscribe<GoldChangedEvent>(OnGoldChanged, this);
            _eventBus.Subscribe<ConscienceChangedEvent>(OnConscienceChanged, this);
            _eventBus.Subscribe<WaveStartEvent>(OnWaveStart, this);
            _eventBus.Subscribe<BossDamagedEvent>(OnBossDamaged, this);
            
            // 查找老板
            _boss = FindObjectOfType<Boss>();
            
            // 初始化 UI
            RefreshAll();
            
            // 隐藏选中面板
            if (_selectedPanel != null)
            {
                _selectedPanel.SetActive(false);
            }
        }
        
        protected override void OnHide()
        {
            _eventBus?.UnsubscribeAll(this);
        }
        
        void Update()
        {
            // 更新选中员工信息
            UpdateSelectedEmployee();
            
            // 更新老板血条
            UpdateBossHP();
        }
        
        // ==================== 刷新 UI ====================
        
        private void RefreshAll()
        {
            RefreshGold();
            RefreshConscience();
            RefreshWave();
            UpdateBossHP();
        }
        
        private void RefreshGold()
        {
            if (_goldText != null && GameManager.Instance != null)
            {
                _goldText.text = $"资金: {GameManager.Instance.Gold}";
            }
        }
        
        private void RefreshConscience()
        {
            if (_conscienceText != null && GameManager.Instance != null)
            {
                _conscienceText.text = $"良心: {GameManager.Instance.Conscience}";
            }
        }
        
        private void RefreshWave()
        {
            if (_waveText != null && GameManager.Instance != null)
            {
                _waveText.text = $"波次: {GameManager.Instance.CurrentWave}";
            }
        }
        
        private void UpdateBossHP()
        {
            if (_boss == null) return;
            
            if (_bossHPBar != null)
            {
                _bossHPBar.value = _boss.Stats.HPPercent;
            }
            
            if (_bossHPText != null)
            {
                _bossHPText.text = $"{_boss.Stats.CurrentHP}/{_boss.Stats.MaxHP}";
            }
        }
        
        private void UpdateSelectedEmployee()
        {
            // 检查老板选中的员工
            if (_boss != null && _boss.SelectedEmployee != null)
            {
                if (_selectedEmployee != _boss.SelectedEmployee)
                {
                    _selectedEmployee = _boss.SelectedEmployee;
                    ShowSelectedEmployee(_selectedEmployee);
                }
                
                // 更新血条
                if (_selectedHPBar != null)
                {
                    _selectedHPBar.value = _selectedEmployee.Stats.HPPercent;
                }
            }
            else if (_selectedEmployee != null)
            {
                _selectedEmployee = null;
                HideSelectedEmployee();
            }
        }
        
        private void ShowSelectedEmployee(Employee employee)
        {
            if (_selectedPanel == null) return;
            
            _selectedPanel.SetActive(true);
            
            if (_selectedNameText != null)
            {
                _selectedNameText.text = employee.Config?.unitName ?? "员工";
            }
            
            if (_selectedLevelText != null)
            {
                _selectedLevelText.text = $"Lv.{employee.Level}";
            }
            
            if (_selectedHPBar != null)
            {
                _selectedHPBar.value = employee.Stats.HPPercent;
            }
            
            // 升级按钮
            UpdateUpgradeButton(employee);
        }
        
        private void HideSelectedEmployee()
        {
            if (_selectedPanel != null)
            {
                _selectedPanel.SetActive(false);
            }
        }
        
        private void UpdateUpgradeButton(Employee employee)
        {
            if (_upgradeButton == null) return;
            
            if (employee.IsMaxLevel)
            {
                _upgradeButton.interactable = false;
                if (_upgradeCostText != null)
                {
                    _upgradeCostText.text = "满级";
                }
            }
            else
            {
                int cost = employee.GetUpgradeCost();
                _upgradeButton.interactable = GameManager.Instance.Gold >= cost;
                
                if (_upgradeCostText != null)
                {
                    _upgradeCostText.text = $"升级 ({cost})";
                }
            }
        }
        
        // ==================== 事件处理 ====================
        
        private void OnGoldChanged(ref GoldChangedEvent evt)
        {
            RefreshGold();
            
            // 显示变化提示
            string sign = evt.Delta > 0 ? "+" : "";
            SetStateText($"资金 {sign}{evt.Delta}");
        }
        
        private void OnConscienceChanged(ref ConscienceChangedEvent evt)
        {
            RefreshConscience();
        }
        
        private void OnWaveStart(ref WaveStartEvent evt)
        {
            RefreshWave();
            SetStateText($"波次 {evt.WaveNumber} 开始！");
        }
        
        private void OnBossDamaged(ref BossDamagedEvent evt)
        {
            UpdateBossHP();
        }
        
        // ==================== 按钮事件 ====================
        
        private void OnUpgradeClicked()
        {
            if (_selectedEmployee != null && !_selectedEmployee.IsMaxLevel)
            {
                if (_selectedEmployee.LevelUp())
                {
                    UpdateUpgradeButton(_selectedEmployee);
                    
                    if (_selectedLevelText != null)
                    {
                        _selectedLevelText.text = $"Lv.{_selectedEmployee.Level}";
                    }
                }
            }
        }
        
        // ==================== 状态提示 ====================
        
        private void SetStateText(string text)
        {
            if (_stateText != null)
            {
                _stateText.text = text;
                CancelInvoke(nameof(ClearStateText));
                Invoke(nameof(ClearStateText), 2f);
            }
        }
        
        private void ClearStateText()
        {
            if (_stateText != null)
            {
                _stateText.text = "";
            }
        }
    }
}
```

## 📝 第二步：创建 HUD 预制体

### 2.1 创建 Canvas

1. 在 Hierarchy 右键 > `UI > Canvas`
2. 设置 Canvas Scaler：
   - UI Scale Mode: `Scale With Screen Size`
   - Reference Resolution: `1920 x 1080`
   - Match: `0.5`

### 2.2 创建 HUD 元素

在 Canvas 下创建：

**资源显示（左上角）**：
```
Canvas
└── ResourcePanel (Empty, 锚点左上)
    ├── GoldText (Text: "资金: 100")
    ├── ConscienceText (Text: "良心: 50")
    └── WaveText (Text: "波次: 1")
```

**老板血条（顶部中央）**：
```
Canvas
└── BossHPPanel (Empty, 锚点顶部中央)
    ├── BossHPBar (Slider)
    └── BossHPText (Text: "100/100")
```

**选中员工面板（左下角）**：
```
Canvas
└── SelectedPanel (Empty, 锚点左下，默认隐藏)
    ├── SelectedNameText (Text: "员工名")
    ├── SelectedLevelText (Text: "Lv.1")
    ├── SelectedHPBar (Slider)
    └── UpgradeButton (Button)
        └── UpgradeCostText (Text: "升级 (50)")
```

**状态提示（底部中央）**：
```
Canvas
└── StateText (Text, 锚点底部中央)
```

### 2.3 挂载脚本并绑定

1. 在 Canvas 上添加 `GameHUDPanel` 脚本
2. 把各个 UI 元素拖到对应字段

### 2.4 保存为预制体

1. 在 `Assets/_Game/Resources/UI/Panels` 文件夹下保存
2. 命名为 `GameHUDPanel`
3. 删除场景中的 Canvas（预制体已保存）

## 📝 第三步：修改 GameManager 打开 HUD

### 3.1 更新 GameManager

在 `GameManager.cs` 的 `Start` 方法中添加：

```csharp
void Start()
{
    // ... 原有代码 ...
    
    // 获取 UI 管理器并打开 HUD
    var uiManager = ServiceLocator.Get<UIManager>();
    if (uiManager != null)
    {
        uiManager.Open<GameHUDPanel>();
    }
}
```

需要添加 using：
```csharp
using CYFramework.Modules.UI;
using LiangXin.UI;
```

## 📝 第四步：创建简易 HUD（不用预制体的方式）

如果你觉得预制体太麻烦，这里提供一个直接用代码创建 UI 的简易方案：

### 4.1 创建 SimpleHUD 脚本

```csharp
using UnityEngine;
using UnityEngine.UI;
using CYFramework.Infrastructure;
using CYFramework.Core.Event;
using LiangXin.Events;
using LiangXin.Unit;

namespace LiangXin.UI
{
    /// <summary>
    /// 简易 HUD（代码创建 UI）
    /// </summary>
    public class SimpleHUD : MonoBehaviour
    {
        private Text _goldText;
        private Text _waveText;
        private Text _stateText;
        private Slider _bossHPBar;
        
        private EventBus _eventBus;
        private Boss _boss;
        private Canvas _canvas;
        
        void Start()
        {
            _eventBus = ServiceLocator.Get<EventBus>();
            
            // 创建 Canvas
            CreateCanvas();
            
            // 创建 UI 元素
            CreateUI();
            
            // 订阅事件
            _eventBus.Subscribe<GoldChangedEvent>(OnGoldChanged, this);
            _eventBus.Subscribe<WaveStartEvent>(OnWaveStart, this);
            
            // 找老板
            _boss = FindObjectOfType<Boss>();
            
            // 初始刷新
            RefreshAll();
        }
        
        void Update()
        {
            // 更新老板血条
            if (_boss != null && _bossHPBar != null)
            {
                _bossHPBar.value = _boss.Stats.HPPercent;
            }
        }
        
        void OnDestroy()
        {
            _eventBus?.UnsubscribeAll(this);
        }
        
        private void CreateCanvas()
        {
            // 创建 Canvas
            var canvasObj = new GameObject("SimpleHUDCanvas");
            _canvas = canvasObj.AddComponent<Canvas>();
            _canvas.renderMode = RenderMode.ScreenSpaceOverlay;
            _canvas.sortingOrder = 100;
            
            canvasObj.AddComponent<CanvasScaler>();
            canvasObj.AddComponent<GraphicRaycaster>();
        }
        
        private void CreateUI()
        {
            // 资金文本（左上）
            _goldText = CreateText("GoldText", new Vector2(10, -10), 
                new Vector2(200, 30), TextAnchor.UpperLeft, "资金: 100");
            
            // 波次文本（左上，资金下方）
            _waveText = CreateText("WaveText", new Vector2(10, -50), 
                new Vector2(200, 30), TextAnchor.UpperLeft, "波次: 0");
            
            // 状态提示（底部中央）
            _stateText = CreateText("StateText", new Vector2(0, 50), 
                new Vector2(400, 50), TextAnchor.LowerCenter, "");
            _stateText.rectTransform.anchorMin = new Vector2(0.5f, 0);
            _stateText.rectTransform.anchorMax = new Vector2(0.5f, 0);
            _stateText.fontSize = 24;
            _stateText.color = Color.yellow;
            
            // 老板血条（顶部中央）
            _bossHPBar = CreateSlider("BossHPBar", new Vector2(0, -30), new Vector2(300, 20));
        }
        
        private Text CreateText(string name, Vector2 position, Vector2 size, 
            TextAnchor alignment, string content)
        {
            var obj = new GameObject(name);
            obj.transform.SetParent(_canvas.transform, false);
            
            var text = obj.AddComponent<Text>();
            text.text = content;
            text.font = Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");
            text.fontSize = 20;
            text.color = Color.white;
            text.alignment = alignment;
            
            var rect = text.rectTransform;
            rect.anchorMin = new Vector2(0, 1);
            rect.anchorMax = new Vector2(0, 1);
            rect.pivot = new Vector2(0, 1);
            rect.anchoredPosition = position;
            rect.sizeDelta = size;
            
            // 添加阴影
            var shadow = obj.AddComponent<Shadow>();
            shadow.effectColor = Color.black;
            
            return text;
        }
        
        private Slider CreateSlider(string name, Vector2 position, Vector2 size)
        {
            var obj = new GameObject(name);
            obj.transform.SetParent(_canvas.transform, false);
            
            var slider = obj.AddComponent<Slider>();
            slider.minValue = 0;
            slider.maxValue = 1;
            slider.value = 1;
            
            var rect = slider.GetComponent<RectTransform>();
            rect.anchorMin = new Vector2(0.5f, 1);
            rect.anchorMax = new Vector2(0.5f, 1);
            rect.pivot = new Vector2(0.5f, 1);
            rect.anchoredPosition = position;
            rect.sizeDelta = size;
            
            // 背景
            var bg = new GameObject("Background");
            bg.transform.SetParent(obj.transform, false);
            var bgImage = bg.AddComponent<Image>();
            bgImage.color = Color.gray;
            var bgRect = bg.GetComponent<RectTransform>();
            bgRect.anchorMin = Vector2.zero;
            bgRect.anchorMax = Vector2.one;
            bgRect.sizeDelta = Vector2.zero;
            
            // 填充
            var fill = new GameObject("Fill");
            fill.transform.SetParent(obj.transform, false);
            var fillImage = fill.AddComponent<Image>();
            fillImage.color = Color.green;
            var fillRect = fill.GetComponent<RectTransform>();
            fillRect.anchorMin = Vector2.zero;
            fillRect.anchorMax = new Vector2(1, 1);
            fillRect.sizeDelta = Vector2.zero;
            
            slider.fillRect = fillRect;
            
            return slider;
        }
        
        private void RefreshAll()
        {
            if (GameManager.Instance != null)
            {
                _goldText.text = $"资金: {GameManager.Instance.Gold}";
                _waveText.text = $"波次: {GameManager.Instance.CurrentWave}";
            }
        }
        
        private void OnGoldChanged(ref GoldChangedEvent evt)
        {
            _goldText.text = $"资金: {evt.NewAmount}";
            
            string sign = evt.Delta > 0 ? "+" : "";
            ShowState($"资金 {sign}{evt.Delta}");
        }
        
        private void OnWaveStart(ref WaveStartEvent evt)
        {
            _waveText.text = $"波次: {evt.WaveNumber}";
            ShowState($"波次 {evt.WaveNumber} 开始！");
        }
        
        private void ShowState(string text)
        {
            _stateText.text = text;
            CancelInvoke(nameof(ClearState));
            Invoke(nameof(ClearState), 2f);
        }
        
        private void ClearState()
        {
            _stateText.text = "";
        }
    }
}
```

### 4.2 使用简易 HUD

1. 在 `[Game]` 对象上添加 `SimpleHUD` 脚本
2. 运行游戏即可看到 UI
